cmake_minimum_required(VERSION 3.24)
project(JerzyAudioPlatform VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JACK REQUIRED IMPORTED_TARGET jack)
add_executable(jerzy-audio-core core/main.cpp)
target_link_libraries(jerzy-audio-core PRIVATE PkgConfig::JACK Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(jerzy-audio-core PRIVATE -Wall -Wextra -Wpedantic)
foreach(module utility-gain-meter bass-ambient big-muff tape-drive cab-ir)
 add_library(${module} MODULE modules/${module}/module.cpp)
 target_include_directories(${module} PRIVATE module-sdk)
 set_target_properties(${module} PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/modules/${module}")
 configure_file(modules/${module}/manifest.json "${CMAKE_BINARY_DIR}/modules/${module}/manifest.json" COPYONLY)
 install(TARGETS ${module} LIBRARY DESTINATION modules/${module})
 install(FILES modules/${module}/manifest.json DESTINATION modules/${module})
endforeach()
install(TARGETS jerzy-audio-core RUNTIME DESTINATION bin)
install(DIRECTORY web-ui/ DESTINATION web-ui)
install(DIRECTORY module-manager/ DESTINATION module-manager)
install(FILES module-sdk/jerzy_module.h DESTINATION module-sdk)
if(BUILD_TESTING)

option(JERZY_BUILD_ORIGINALS "Build complete original DSP modules" ON)
if(JERZY_BUILD_ORIGINALS)
 set(JUCE_MODULES_ONLY ON CACHE BOOL "" FORCE)
 set(JERZY_JUCE_ROOT "" CACHE PATH "Pinned JUCE 7.0.12 checkout")
 if(JERZY_JUCE_ROOT)
  add_subdirectory("${JERZY_JUCE_ROOT}" juce-original)
 else()
  include(FetchContent)
  FetchContent_Declare(juce_original URL https://github.com/juce-framework/JUCE/archive/4f43011b96eb0636104cb3e433894cda98243626.tar.gz URL_HASH SHA256=efaeeed2ca988d3ade7b21dd01dca5c79fd9a2fb5884ea6140b31a7a87e61053 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(juce_original)
  set(JERZY_JUCE_ROOT "${juce_original_SOURCE_DIR}")
 endif()
 add_library(jerzy-juce-dsp STATIC)
 set_target_properties(jerzy-juce-dsp PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
 target_link_libraries(jerzy-juce-dsp PRIVATE juce::juce_dsp)
 target_include_directories(jerzy-juce-dsp PUBLIC "${JERZY_JUCE_ROOT}/modules")
 target_compile_definitions(jerzy-juce-dsp PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0 JUCE_STANDALONE_APPLICATION=1 JUCE_USE_FLAC=0 JUCE_USE_OGGVORBIS=0 JUCE_USE_MP3AUDIOFORMAT=0)
 foreach(module tape-drive-original small-clone-original atomic-cluster-original memory-delay-original mx-delay-original big-muff-original bass-ambient-original bass-ambient-pro-original)
  add_library(${module} MODULE modules/${module}/module.cpp)
  target_include_directories(${module} PRIVATE module-sdk)
  target_compile_options(${module} PRIVATE -fno-math-errno -fno-trapping-math)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
   target_compile_options(${module} PRIVATE -mtune=cortex-a72)
  endif()
  set_target_properties(${module} PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/modules/${module}")
  configure_file(modules/${module}/manifest.json "${CMAKE_BINARY_DIR}/modules/${module}/manifest.json" COPYONLY)
 endforeach()
 target_link_libraries(big-muff-original PRIVATE jerzy-juce-dsp)
endif()
