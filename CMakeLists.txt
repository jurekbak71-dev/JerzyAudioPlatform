cmake_minimum_required(VERSION 3.24)
project(JerzyAudioPlatform VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JACK REQUIRED IMPORTED_TARGET jack)
add_executable(jerzy-audio-core core/main.cpp)
target_link_libraries(jerzy-audio-core PRIVATE PkgConfig::JACK Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(jerzy-audio-core PRIVATE -Wall -Wextra -Wpedantic)
foreach(module utility-gain-meter bass-ambient big-muff tape-drive cab-ir)
 add_library(${module} MODULE modules/${module}/module.cpp)
 target_include_directories(${module} PRIVATE module-sdk)
 set_target_properties(${module} PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/modules/${module}")
 configure_file(modules/${module}/manifest.json "${CMAKE_BINARY_DIR}/modules/${module}/manifest.json" COPYONLY)
 install(TARGETS ${module} LIBRARY DESTINATION modules/${module})
 install(FILES modules/${module}/manifest.json DESTINATION modules/${module})
endforeach()
install(TARGETS jerzy-audio-core RUNTIME DESTINATION bin)
install(DIRECTORY web-ui/ DESTINATION web-ui)
install(DIRECTORY module-manager/ DESTINATION module-manager)
install(FILES module-sdk/jerzy_module.h DESTINATION module-sdk)
if(BUILD_TESTING)
 add_executable(module-tests tests/modules.cpp)
 target_include_directories(module-tests PRIVATE module-sdk)
 target_link_libraries(module-tests PRIVATE ${CMAKE_DL_LIBS})
 target_compile_options(module-tests PRIVATE -UNDEBUG)
 add_test(NAME ModulyDSP COMMAND module-tests ${CMAKE_BINARY_DIR}/modules)
endif()
if(BUILD_TESTING)
 add_executable(jack-probe tests/jack_probe.cpp)
 target_link_libraries(jack-probe PRIVATE PkgConfig::JACK Threads::Threads)
endif()
